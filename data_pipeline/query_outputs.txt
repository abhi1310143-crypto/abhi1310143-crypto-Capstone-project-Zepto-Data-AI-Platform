Query 5 - IN and JOIN

           SELECT
                b.title,
                b.rating,
                b.price_inr,
                c.category_name
            FROM books AS b
            JOIN categories AS c ON b.category_id = c.category_id
            WHERE b.rating IN (4, 5)
            ORDER BY b.rating DESC
            LIMIT 10
